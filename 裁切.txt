(function (Scratch) {
  'use strict';

  const vm = Scratch.vm;

  /**
   * 讀取圖片（支援 CORS）
   * @param {string} url
   * @returns {Promise<HTMLImageElement>}
   */
  function loadImage(url) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      // 重要：若來源伺服器允許 CORS，這樣畫到 canvas 才不會被汙染
      img.crossOrigin = 'anonymous';
      img.onload = () => resolve(img);
      img.onerror = (e) => reject(new Error('圖片載入失敗：' + url));
      img.src = String(url);
    });
  }

  /**
   * 依格式產生 MIME type
   */
  function fmtToMime(fmt) {
    const f = String(fmt || 'png').toLowerCase();
    if (f === 'jpeg' || f === 'jpg') return 'image/jpeg';
    if (f === 'webp') return 'image/webp';
    return 'image/png';
  }

  /**
   * 安全數字轉換
   */
  function toNum(v, def = 0) {
    const n = Number(v);
    return Number.isFinite(n) ? n : def;
  }

  class CropURL {
    getInfo() {
      return {
        id: 'cropURL',
        name: '裁切URL',
        color1: '#2f80ed',
        color2: '#1c6dd0',
        color3: '#185fb6',
        blocks: [
          {
            opcode: 'crop',
            blockType: Scratch.BlockType.REPORTER,
            text: '裁切URL [URL] x [X] y [Y] 寬 [W] 高 [H] 單位 [UNIT] 格式 [FMT] 品質 [Q]',
            arguments: {
              URL: { type: Scratch.ArgumentType.STRING, defaultValue: 'https://extensions.turbowarp.org/dango.png' },
              X: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              Y: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              W: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              H: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              UNIT: { type: Scratch.ArgumentType.STRING, menu: 'units', defaultValue: 'px' },
              FMT: { type: Scratch.ArgumentType.STRING, menu: 'fmts', defaultValue: 'png' },
              Q: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0.92 }
            }
          },
          {
            opcode: 'cropPercent',
            blockType: Scratch.BlockType.REPORTER,
            text: '裁切URL(%) [URL] 左% [LX] 上% [TY] 寬% [PW] 高% [PH] 格式 [FMT] 品質 [Q]',
            arguments: {
              URL: { type: Scratch.ArgumentType.STRING, defaultValue: 'https://extensions.turbowarp.org/dango.png' },
              LX: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              TY: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              PW: { type: Scratch.ArgumentType.NUMBER, defaultValue: 50 },
              PH: { type: Scratch.ArgumentType.NUMBER, defaultValue: 50 },
              FMT: { type: Scratch.ArgumentType.STRING, menu: 'fmts', defaultValue: 'png' },
              Q: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0.92 }
            }
          },
          {
            opcode: 'resizeAndCrop',
            blockType: Scratch.BlockType.REPORTER,
            text: '裁切+縮放URL [URL] x [X] y [Y] 寬 [W] 高 [H] 輸出寬 [OW] 高 [OH] 格式 [FMT] 品質 [Q]',
            arguments: {
              URL: { type: Scratch.ArgumentType.STRING, defaultValue: 'https://extensions.turbowarp.org/dango.png' },
              X: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              Y: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
              W: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              H: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              OW: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              OH: { type: Scratch.ArgumentType.NUMBER, defaultValue: 100 },
              FMT: { type: Scratch.ArgumentType.STRING, menu: 'fmts', defaultValue: 'png' },
              Q: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0.92 }
            }
          }
        ],
        menus: {
          units: {
            acceptReporters: true,
            items: [
              { text: '像素(px)', value: 'px' },
              { text: '百分比(%)', value: '%' }
            ]
          },
          fmts: {
            acceptReporters: true,
            items: [
              { text: 'PNG', value: 'png' },
              { text: 'JPEG', value: 'jpeg' },
              { text: 'WebP', value: 'webp' }
            ]
          }
        }
      };
    }

    async crop(args) {
      const url = String(args.URL || '');
      const X = toNum(args.X, 0);
      const Y = toNum(args.Y, 0);
      const W = Math.max(1, toNum(args.W, 1));
      const H = Math.max(1, toNum(args.H, 1));
      const unit = String(args.UNIT || 'px').toLowerCase();
      const fmt = fmtToMime(args.FMT);
      const q = Math.max(0, Math.min(1, toNum(args.Q, 0.92)));
      try {
        const img = await loadImage(url);
        let sx = X, sy = Y, sw = W, sh = H;
        if (unit === '%') {
          sx = img.width * (X / 100);
          sy = img.height * (Y / 100);
          sw = img.width * (W / 100);
          sh = img.height * (H / 100);
        }
        // 夾在範圍內
        sx = Math.max(0, Math.min(img.width, sx));
        sy = Math.max(0, Math.min(img.height, sy));
        sw = Math.max(1, Math.min(img.width - sx, sw));
        sh = Math.max(1, Math.min(img.height - sy, sh));

        const canvas = document.createElement('canvas');
        canvas.width = Math.round(sw);
        canvas.height = Math.round(sh);
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);

        // PNG 無視品質參數；JPEG/WebP 使用 q
        const dataURL = (fmt === 'image/png')
          ? canvas.toDataURL(fmt)
          : canvas.toDataURL(fmt, q);
        return dataURL;
      } catch (e) {
        return '' + e.message;
      }
    }

    async cropPercent(args) {
      // 百分比版本：直接轉成 crop 所需參數
      return this.crop({
        URL: args.URL,
        X: args.LX,
        Y: args.TY,
        W: args.PW,
        H: args.PH,
        UNIT: '%',
        FMT: args.FMT,
        Q: args.Q
      });
    }

    async resizeAndCrop(args) {
      const url = String(args.URL || '');
      const X = toNum(args.X, 0);
      const Y = toNum(args.Y, 0);
      const W = Math.max(1, toNum(args.W, 1));
      const H = Math.max(1, toNum(args.H, 1));
      const OW = Math.max(1, toNum(args.OW, W));
      const OH = Math.max(1, toNum(args.OH, H));
      const fmt = fmtToMime(args.FMT);
      const q = Math.max(0, Math.min(1, toNum(args.Q, 0.92)));

      try {
        const img = await loadImage(url);
        let sx = Math.max(0, Math.min(img.width, X));
        let sy = Math.max(0, Math.min(img.height, Y));
        let sw = Math.max(1, Math.min(img.width - sx, W));
        let sh = Math.max(1, Math.min(img.height - sy, H));

        const canvas = document.createElement('canvas');
        canvas.width = Math.round(OW);
        canvas.height = Math.round(OH);
        const ctx = canvas.getContext('2d');
        ctx.imageSmoothingEnabled = true;
        ctx.imageSmoothingQuality = 'high';
        ctx.drawImage(img, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);

        const dataURL = (fmt === 'image/png')
          ? canvas.toDataURL(fmt)
          : canvas.toDataURL(fmt, q);
        return dataURL;
      } catch (e) {
        return '' + e.message;
      }
    }
  }

  Scratch.extensions.register(new CropURL());
})(Scratch);
